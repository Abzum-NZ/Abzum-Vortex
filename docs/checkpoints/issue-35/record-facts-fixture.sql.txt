-- Fixed read fact loader. It accepts one record identity and nothing else.
create function vortex_access.test_load_exact_record_read_facts(
  p_record_id uuid
)
returns table (
  fact_role text,
  operation_key text,
  action_kind text,
  permission_id uuid,
  release_id uuid,
  storage_scope text,
  module_root_id uuid,
  record_type_id uuid,
  storage_contract_id uuid,
  record_id uuid,
  ownership_mode text,
  owner_organization_account_id uuid,
  owner_group_id uuid,
  ownership_witness boolean,
  condition_witness boolean,
  share_contribution_count bigint,
  related_record_ids uuid[]
)
language plpgsql
volatile
security definer
set search_path = ''
as $function$
declare
  context_value jsonb;
  declaration_row record;
  binding_row record;
  target_row record;
  checked_at timestamptz;
begin
  context_value := vortex_access.validated_human_request_context();
  checked_at := pg_catalog.clock_timestamp();

  select * into strict declaration_row
  from vortex_access.test_exact_record_operation_declarations as fixed
  where fixed.operation_key = 'record.read';

  select * into binding_row
  from vortex_access.test_exact_record_release_bindings as installed
  where installed.organization_id = (context_value ->> 'organizationId')::uuid
    and installed.application_root_id =
      (context_value ->> 'applicationRootId')::uuid
    and installed.record_type_id = declaration_row.record_type_id
    and installed.release_state = 'current';
  if not found or binding_row.release_id <> declaration_row.release_id then
    raise exception using errcode = '22023',
      message = 'Exact record release binding is unavailable';
  end if;

  select * into target_row
  from vortex_access.test_exact_record_rows as stored
  where stored.record_id = p_record_id
    and stored.lifecycle_state = 'active';
  if not found then
    raise exception using errcode = '22023',
      message = 'Exact record target facts are unavailable';
  end if;
  if target_row.record_scope ->> 'storageScope'
      is distinct from binding_row.storage_scope
    or (target_row.record_scope ->> 'organizationId')::uuid
      is distinct from binding_row.organization_id
    or (target_row.record_scope ->> 'applicationRootId')::uuid
      is distinct from case
        when binding_row.storage_scope = 'application_contained'
        then binding_row.application_root_id else null end
    or (target_row.record_scope ->> 'moduleRootId')::uuid
      is distinct from binding_row.module_root_id
    or (target_row.record_scope ->> 'recordTypeId')::uuid
      is distinct from binding_row.record_type_id
    or (target_row.record_scope ->> 'storageContractId')::uuid
      is distinct from binding_row.storage_contract_id
    or (target_row.record_scope ->> 'recordId')::uuid is distinct from p_record_id
  then
    raise exception using errcode = '22023',
      message = 'Exact record facts do not match the installed storage binding';
  end if;

  return query
  select
    'target'::text, declaration_row.operation_key, declaration_row.action_kind,
    declaration_row.permission_id, binding_row.release_id,
    binding_row.storage_scope, binding_row.module_root_id,
    binding_row.record_type_id, binding_row.storage_contract_id, p_record_id,
    target_row.ownership_mode, target_row.owner_organization_account_id,
    target_row.owner_group_id, visibility.admitted,
    vortex_access.evaluate_permission_saved_condition(
      declaration_row.permission_record_scope,
      binding_row.saved_condition_release, binding_row.record_type_shape,
      target_row.field_values,
      (context_value ->> 'organizationAccountId')::uuid
    ),
    (
      select pg_catalog.count(*)
      from vortex_access.read_current_direct_record_share_contributions(
        binding_row.organization_id, binding_row.application_root_id,
        binding_row.module_root_id, binding_row.record_type_id,
        binding_row.storage_contract_id, binding_row.storage_scope,
        p_record_id, (context_value ->> 'organizationId')::uuid,
        (context_value ->> 'applicationRootId')::uuid,
        (context_value ->> 'organizationAccountId')::uuid, checked_at
      ) as contribution
    ),
    (
      select pg_catalog.array_agg(related.record_id order by related.record_id)
      from vortex_access.test_exact_record_relationship_edges as edge
      join vortex_access.test_exact_record_rows as related
        on related.record_id = edge.to_record_id
        and related.lifecycle_state = 'active'
      join vortex_access.test_exact_record_relationship_declarations as route
        on route.relationship_id = edge.relationship_id
        and route.release_id = binding_row.release_id

        'configuration', routine.proconfig
      )
      order by routine.proname collate "C"
    )
    from pg_catalog.pg_proc as routine
    join pg_catalog.pg_namespace as namespace
      on namespace.oid = routine.pronamespace
    join pg_catalog.pg_roles as owner_role
      on owner_role.oid = routine.proowner
    where namespace.nspname = 'vortex_access'
      and routine.proname like 'test_load_exact_record_%_facts'
  ),
  (
    select pg_catalog.jsonb_agg(
      pg_catalog.jsonb_build_object(
        'name', expected.name, 'owner', 'postgres',
        'securityDefiner', true, 'configuration', array['search_path=""']
      )
      order by expected.name collate "C"
    )
    from (values
      ('test_load_exact_record_create_facts'),
      ('test_load_exact_record_delete_facts'),
      ('test_load_exact_record_read_facts'),
      ('test_load_exact_record_update_facts')
    ) as expected(name)
  ),
  'each loader is owner-held, definer-rights and empty-search-path'
);
